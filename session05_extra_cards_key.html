Letter grades

Before you run: return ends the function. 95 passes the first gate, the function returns "A", and the gate for 80 never runs.

This is the program with the Go further part: count_grade and the last two lines.

def grade_of(score):
    if score >= 90:
        return "A"
    if score >= 80:
        return "B"
    if score >= 70:
        return "C"
    return "not yet"

def count_grade(scores, letter):
    count = 0
    for s in scores:
        if grade_of(s) == letter:
            count = count + 1
    return count

print(grade_of(95))
print(grade_of(80))
print(grade_of(42))

scores = [72, 95, 88, 97, 67]
for s in scores:
    grade = grade_of(s)
    print(f"{s} gets {grade}")

print(count_grade(scores, "A"))

Card B · build it from the steps · about 20 minutes

Dice game

Before you run: the lowest is 2 and the highest is 12.

import random

def roll():
    return random.choice([1, 2, 3, 4, 5, 6])

def roll_two():
    return roll() + roll()

def winner_of(you, computer):
    if you > computer:
        return "you"
    if computer > you:
        return "computer"
    return "tie"

your_wins = 0
for round_number in range(1, 6):
    input("Press Enter to roll.")
    you = roll_two()
    computer = roll_two()
    result = winner_of(you, computer)
    print(f"Round {round_number}: you rolled {you}, the computer rolled {computer}. Winner: {result}")
    if result == "you":
        your_wins = your_wins + 1

print(f"You won {your_wins} of 5 rounds.")

Go further:

def roll_many(count):
    total = 0
    for i in range(count):
        total = total + roll()
    return total

Ten rolls of 1 add up to 10 and ten rolls of 6 add up to 60, so roll_many(10) is always between 10 and 60.

The fixed program:

def total_of(numbers):
    total = 0
    for n in numbers:
        total = total + n
    return total

def average_of(numbers):
    return total_of(numbers) / len(numbers)

print(average_of([12, 20, 28, 20]))
